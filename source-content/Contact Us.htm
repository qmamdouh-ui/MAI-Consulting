const EMAILJS_PUBLIC_KEY  = 'JJs3lu55OHI_pbjth';

const EMAILJS_SERVICE_ID  = 'service_an94235';

const EMAILJS_TEMPLATE_ID = 'template_bv6m1l3';

 

/* initialise EmailJS */

emailjs.init({ publicKey: EMAILJS_PUBLIC_KEY });

 

/* ── FORM SUBMISSION ── */

document.getElementById('contact-form').addEventListener('submit', function(e) {

  e.preventDefault();

 

  /* hide previous status */

  document.getElementById('msg-success').classList.remove('visible');

  document.getElementById('msg-error').classList.remove('visible');

 

  /* basic client-side validation */

  var name    = document.getElementById('from_name').value.trim();

  var email   = document.getElementById('from_email').value.trim();

  var sector  = document.getElementById('sector').value;

  var service = document.getElementById('service').value;

  var message = document.getElementById('message').value.trim();

 

  if (!name || !email || !sector || !service || !message) {

    showError('Please fill in all required fields before submitting.');

    return;

  }

  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {

    showError('Please enter a valid email address.');

    return;

  }

 

  /* collect preferred contact methods */

  var preferred = [];

  document.querySelectorAll('input[type=checkbox]:checked').forEach(function(cb) {

    preferred.push(cb.value);

  });

 

  /* build template params */

  var params = {

    from_name:    name,

    from_email:   email,

    organization: document.getElementById('organization').value.trim() || '—',

    job_title:    document.getElementById('job_title').value.trim()    || '—',

    sector:       sector,

    team_size:    document.getElementById('team_size').value           || '—',

    service:      service,

    referral:     document.getElementById('referral').value            || '—',

    preferred:    preferred.length ? preferred.join(', ') : '—',

    message:      message,

    to_email:     'q.mamdouh@mai4consulting.com'

  };

 

  /* loading state */

  setLoading(true);

 

  emailjs.send(EMAILJS_SERVICE_ID, EMAILJS_TEMPLATE_ID, params)

    .then(function() {

      setLoading(false);

      document.getElementById('msg-success').classList.add('visible');

      document.getElementById('contact-form').reset();

      document.getElementById('msg-success').scrollIntoView({behavior:'smooth', block:'nearest'});

    })

    .catch(function(err) {

      setLoading(false);

      var detail = err && err.text ? ' (' + err.text + ')' : '';

      showError('Something went wrong' + detail + '. Please try again or email q.mamdouh@mai4consulting.com directly.');

      console.error('EmailJS error:', err);

    });

});

 

function setLoading(on) {

  var btn     = document.getElementById('submit-btn');

  var txt     = document.getElementById('btn-text');

  var icon    = document.getElementById('btn-icon');

  var spinner = document.getElementById('btn-spinner');

  btn.disabled   = on;

  txt.textContent = on ? 'Sending…' : 'Send Message';

  icon.style.display    = on ? 'none' : '';

  spinner.style.display = on ? 'block' : 'none';

}

 

function showError(msg) {

  document.getElementById('error-text').textContent = msg;

  var el = document.getElementById('msg-error');

  el.classList.add('visible');

  el.scrollIntoView({behavior:'smooth', block:'nearest'});

}
