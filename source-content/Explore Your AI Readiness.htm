    //    JSON.parse works server-side regardless of the declared type.

    let finalPdfLink = '';

    try{

      const driveResponse = await fetch(GAS_URL, {

        method: 'POST',

        headers: { 'Content-Type': 'text/plain;charset=utf-8' },

        body: JSON.stringify({ filename: fname, base64: base64Data })

      });

      const driveResult = await driveResponse.json();

      if(driveResult && driveResult.status === 'success' && driveResult.url){

        finalPdfLink = driveResult.url;

        console.log('Drive upload OK:', finalPdfLink);

      } else {

        console.error('Drive upload failed:', driveResult);

        finalPdfLink = 'Upload failed — please contact us directly for your report.';

      }

    }catch(driveErr){

      console.error('Drive upload error:', driveErr);

      finalPdfLink = 'Upload failed — please contact us directly for your report.';

    // 4. Email the notification with the Drive link

    const{dimScores,dimMax}=getDimData();

/* Convert #rrggbb to [r,g,b]. */

function hexRGB(hex){

  const m=/^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex||'#000000');

  return m?[parseInt(m[1],16),parseInt(m[2],16),parseInt(m[3],16)]:[0,0,0];

}
